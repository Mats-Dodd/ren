handler down_every(every: Int) over any =
  var n = 0
  call =>
    n += 1
    if n % every == 0 then call.fail(Down) else try call.forward()

pub fn retry<T>(n: Int, on: Tags, body: fn() -> T) -> T =
  try match body()
    fail on if n > 1 => try retry(n - 1, on: on, body)
    v => v
