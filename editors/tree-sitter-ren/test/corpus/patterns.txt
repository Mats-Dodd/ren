fn arg(a: Str) -> Str =
  match a
    "-h" | "--help" => "help"
    "--out=" ..path => path
    ..stem ".json" => stem
    "<" ..inner ">" => inner
    "--" .. => "flag"
    "a" "b" => "ab"
    _ => a

    [] => 0

    [a, .., z] => 3
    ["v1" | "v2", "keys", k] => 4
